\begin{document}

\begin{center}
\begin{adjustbox}{max width=\textwidth, max totalheight=\textheight}
\begin{tikzpicture}


    % Riga 1
    \umlclass[x=-9, y=0]{Profilo}{
        - biografia : String \\
        - fotoUrl : String
    }{
        + visualizzaProfilo() : void
    }

    \umlclass[x=0, y=0, type=abstract]{Utente}{
        - id : int \\
        - nome : String
    }{
        + cercaUtenti() : List<Utente> \\
        + visualizzaListaEventi() : List<Evento>
    }

    \umlclass[x=9, y=0]{Evento}{
        - idEvento : int \\
        - titolo : String \\
        - data : Date \\
        - descrizione : String
    }{
        + aggiungiEvento() : boolean \\
        + modificaEvento() : boolean \\
        + eliminaEvento() : boolean \\
        + visualizzaDettagli() : Evento
    }

   
    \umlclass[x=-9, y=-6]{Ospite}{}{
        + registra() : UtenteRegistrato
    }

    \umlclass[x=0, y=-6]{UtenteRegistrato}{
        - email : String \\
        - password : String \\
        - tokenSessione : String
    }{
        + login() : boolean \\
        + logout() : void
    }

    \umlclass[x=9, y=-9]{Notifica}{
        - idNotifica : int \\
        - testo : String \\
        - letta : boolean
    }{
        + invia() : void \\
        + segnaComeLetta() : void
    }

    \umlclass[x=-9, y=-12]{Chat}{
        - idChat : int \\
        - messaggi : List<String>
    }{
        + inviaMessaggio() : void \\
        + leggiChat() : void
    }

    \umlclass[x=18, y=-13]{Biglietto}{
        - idBiglietto : int \\
        - prezzo : float \\
        - codiceQR : String
    }{
        + acquista() : boolean \\
        + annulla() : boolean
    }


    \umlinherit{Ospite}{Utente}
    \umlinherit{UtenteRegistrato}{Utente}

    \umlcompo[mult1=1, mult2=1, assoc=possiede]{Utente}{Profilo}

    \umlassoc[mult1=1, mult2=*, assoc=gestisce]{UtenteRegistrato}{Evento}
    \umlassoc[mult1=*, mult2=*, assoc=partecipa a]{UtenteRegistrato}{Chat}
    \umlassoc[mult1=1, mult2=*, assoc=riceve]{UtenteRegistrato}{Notifica}
    \umlassoc[geometry=|-, mult1=1, mult2=*, assoc=acquista]{UtenteRegistrato}{Biglietto}

    \umlassoc[mult1=*, mult2=1, assoc=riferito a]{Biglietto}{Evento}

    \umlassoc[mult1=*, mult2=0..1, assoc=riguarda]{Notifica}{Evento}

\end{tikzpicture}
